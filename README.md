// Extract M-Pesa code from full message or direct code
function extractMpesaCode(input: string): string {
  // Clean input
  const cleanInput = input.trim().toUpperCase();
  
  // If it's already a short code (alphanumeric, ~10 chars), return it
  if (/^[A-Z0-9]{8,12}$/.test(cleanInput)) {
    return cleanInput;
  }
  
  // Try to extract from full M-Pesa message
  // Pattern: Look for codes like "SJK1234ABC" or "UAUPQ586WQ"
  const patterns = [
    /\b([A-Z]{2,4}[A-Z0-9]{6,10})\b/,  // Standard M-Pesa code pattern
    /transaction\s+([A-Z0-9]{8,12})/i,
    /code\s*:?\s*([A-Z0-9]{8,12})/i,
    /\b([A-Z][A-Z0-9]{8,11})\b/,  // Starts with letter
  ];
  
  for (const pattern of patterns) {
    const match = cleanInput.match(pattern);
    if (match && match[1]) {
      return match[1];
    }
  }
  
  // Return original input if no pattern matches
  return cleanInput;
}

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response(null, { headers: corsHeaders });
  }

  try {
    const SUPABASE_URL = Deno.env.get('SUPABASE_URL');
    const SUPABASE_SERVICE_KEY = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY');

    if (!SUPABASE_URL || !SUPABASE_SERVICE_KEY) {
      throw new Error('Supabase credentials not configured');
    }

    const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_KEY);

    const url = new URL(req.url);
    const rawInput = url.searchParams.get('transaction_id') || url.searchParams.get('mpesa_code') || '';

    if (!rawInput) {
      return new Response(
        JSON.stringify({ success: false, error: 'M-Pesa code required' }),
        { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
      );
    }

    // Extract M-Pesa code from input (supports full message)
    const mpesaCode = extractMpesaCode(rawInput);
    console.log('Searching for M-Pesa code:', mpesaCode);

    // Search by mpesa_code first (this is the actual M-Pesa transaction code)
    let { data: payment, error } = await supabase
      .from('payments')
      .select('*, download_tokens(*)')
      .eq('mpesa_code', mpesaCode)
      .single();

    // If not found by mpesa_code, try transaction_id (internal reference)
    if (error || !payment) {
      const result = await supabase
        .from('payments')
        .select('*, download_tokens(*)')
        .eq('transaction_id', mpesaCode)
        .single();
      
      payment = result.data;
      error = result.error;
    }

    if (error || !payment) {
      console.log('Payment not found for code:', mpesaCode);
      return new Response(
        JSON.stringify({ success: false, error: 'Payment not found', status: 'not_found' }),
        { status: 404, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
      );
    }

    console.log('Payment found:', payment.id, 'Status:', payment.payment_status);

    // Find valid download token
    let downloadToken = null;
    if (payment.download_tokens && payment.download_tokens.length > 0) {
      const validTokens = payment.download_tokens.filter(
        (t: any) => !t.is_used && new Date(t.expires_at) > new Date() && t.download_count < t.max_downloads
      );
      if (validTokens.length > 0) {
        downloadToken = validTokens[0].token;
      }
    }

    return new Response(
      JSON.stringify({
        success: true,
        status: payment.payment_status,
        mpesa_code: payment.mpesa_code,
        download_token: downloadToken,
        package_type: payment.package_type || 'full_project',

      }),

function normalizePhoneNumber(phone: string): string {
  phone = phone.trim().replace(/\D+/g, '');
  if (/^254\d{9}$/.test(phone)) return phone;
  if (/^07\d{8}$/.test(phone)) return '254' + phone.substring(1);
  if (/^011\d{7}$/.test(phone)) return '254' + phone.substring(1);
  if (/^\+254\d{9}$/.test(phone)) return phone.substring(1);
  return phone;
}

function generateSecureToken(): string {
  const array = new Uint8Array(32);
  crypto.getRandomValues(array);
  return Array.from(array, byte => byte.toString(16).padStart(2, '0')).join('');
}
